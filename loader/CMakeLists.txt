cmake_minimum_required(VERSION 3.15)
project(AppBoxLoader)

add_executable(${PROJECT_NAME} WIN32
    "rpc/__init__.cpp"
    "rpc/Environment.cpp"
    "rpc/Log.cpp"
    "registry/HiveReader.cpp"
    "utils/CommandLineOptions.cpp"
    "utils/ConvertDosPathToNtPath.cpp"
    "utils/GetExecutableDir.cpp"
    "utils/HiveMerge.cpp"
    "utils/KnownFolder.cpp"
    "utils/MapBaseFS.cpp"
    "utils/Md5.cpp"
    "utils/MiniLauncher.cpp"
    "utils/PatchLayer.cpp"
    "utils/ProcessJob.cpp"
    "utils/WinCall.cpp"
    "widget/MainFrame.cpp"
    "widget/RegistryBrowser.cpp"
    "Loader.cpp"
    "main.cpp"
    "resource.rc"

    # The reader half of the archive layout of the packer. The loader reads
    # the patch packages of the user with it, so a package is extracted with
    # the very same rules the packer writes it with, including the rejection
    # of entry names which would escape the destination folder.
    "${CMAKE_CURRENT_SOURCE_DIR}/../src/core/ZipReader.cpp"
)
appbox_setup_target_wall(${PROJECT_NAME})

target_include_directories(${PROJECT_NAME}
    PRIVATE
        ${CMAKE_CURRENT_SOURCE_DIR}
        ${CMAKE_CURRENT_SOURCE_DIR}/..
)

target_compile_features(${PROJECT_NAME}
    PRIVATE
        cxx_std_17
)

###############################################################################
# Application manifest
#
# resource.rc embeds the wxWidgets application manifest (per monitor v2 DPI
# awareness) through wx/msw/wx.rc. The manifest generated by the linker would be
# a second RT_MANIFEST resource with the same id, which makes cvtres fail with
# CVT1100, so it is turned off here. The wx manifest already carries the
# trustInfo element (level="asInvoker") of the default one.
###############################################################################
if (MSVC)
    target_link_options(${PROJECT_NAME} PRIVATE /MANIFEST:NO)
endif ()

###############################################################################
# Dependency
###############################################################################
if (NOT TARGET AppBoxCommon)
    add_subdirectory(
        ${CMAKE_CURRENT_SOURCE_DIR}/../common
        ${CMAKE_CURRENT_BINARY_DIR}/common
    )
endif ()
target_link_libraries(${PROJECT_NAME} PRIVATE AppBoxCommon)

find_package(base64 REQUIRED)
target_link_libraries(${PROJECT_NAME} PRIVATE base64)

find_package(CLI11 REQUIRED)
target_link_libraries(${PROJECT_NAME} PRIVATE CLI11::CLI11)

find_package(Detours REQUIRED)
target_link_libraries(${PROJECT_NAME} PRIVATE Detours)

find_package(spdlog REQUIRED)
target_link_libraries(${PROJECT_NAME} PRIVATE spdlog::spdlog)
target_compile_options(${PROJECT_NAME} PRIVATE -DSPDLOG_ACTIVE_LEVEL=SPDLOG_LEVEL_TRACE)

find_package(wxWidgets REQUIRED)
target_link_libraries(${PROJECT_NAME} PRIVATE wx::core wx::base)

# Registry access of the sandbox hive browser (RegLoadAppKeyW and friends).
target_link_libraries(${PROJECT_NAME} PRIVATE advapi32)

# Digest of a patch package (BCryptOpenAlgorithmProvider and friends).
target_link_libraries(${PROJECT_NAME} PRIVATE bcrypt)

# Extraction of a patch package. zlib must be configured before libzip: the
# libzip build performs its own find_package(ZLIB 1.1.2 REQUIRED) which reuses
# the already configured target.
find_package(ZLIB REQUIRED)
find_package(libzip REQUIRED)
target_link_libraries(${PROJECT_NAME} PRIVATE libzip::zip)

###############################################################################
# Embed Sandbox
###############################################################################

target_link_libraries(${PROJECT_NAME}
    PRIVATE
        sandbox_resource
)

###############################################################################
# Export for the packer
#
# Copy the built loader below the build resource tree so the packer can embed
# it as a CMakeRC resource (see the AppBox section of the top level
# CMakeLists.txt).
###############################################################################
add_custom_command(TARGET ${PROJECT_NAME} POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/resource/appbox"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "$<TARGET_FILE:${PROJECT_NAME}>"
        "${CMAKE_BINARY_DIR}/resource/appbox/AppBoxLoader.exe"
    COMMENT "Copying AppBoxLoader.exe for the packer resource"
)
