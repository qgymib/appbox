cmake_minimum_required(VERSION 3.15)
project(appbox VERSION 0.1.0)

###############################################################################
# Dependency library form
#
# Every third-party dependency is built as a static library and linked into the
# executables; the sandbox injection modules (AppBoxSandbox32/64.dll) are the
# only dynamic artifacts. The FORCE keyword also heals build trees whose cache
# already holds a stale ON value, for example one created by a third-party
# option() before this guard existed.
###############################################################################
set(BUILD_SHARED_LIBS OFF CACHE BOOL "appbox: build all dependencies as static libraries" FORCE)

# Add cmake search path
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")

###############################################################################
# Runtime library form
#
# The products link the runtime library of the compiler statically as well, so
# they need no Visual C++ Redistributable on the machine which runs them; see
# cmake/StaticRuntime.cmake, which the sandbox project includes as well.
###############################################################################
include(StaticRuntime)

###############################################################################
# Support functions
###############################################################################
function(appbox_setup_target_wall name)
    if (CMAKE_C_COMPILER_ID STREQUAL "MSVC")
        target_compile_options(${name} PRIVATE /W4 /WX)
    else ()
        target_compile_options(${name} PRIVATE -Wall -Wextra -Werror)
    endif ()
endfunction()

###############################################################################
# Build information of the About dialog
#
# The version of the project, the build date, the git revision and the versions
# of the linked third-party libraries are compiled into the binary; the
# application never looks them up at run time. The header which carries them is
# generated by a script which runs on every build, so the values always describe
# the build which produced the executable. The generated header is only touched
# when its content changes, which keeps the translation unit that includes it
# from being recompiled by a build without changes.
###############################################################################
set(APPBOX_GENERATED_DIR "${CMAKE_BINARY_DIR}/generated/appbox")
set(APPBOX_ABOUT_INFO_HEADER "${APPBOX_GENERATED_DIR}/AboutInfo.generated.hpp")

# git is optional: without it the generated header reports `unknown` instead.
find_package(Git QUIET)

add_custom_target(appbox_build_info
    COMMAND ${CMAKE_COMMAND}
        "-DAPPBOX_SOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR}"
        "-DAPPBOX_OUTPUT_FILE=${APPBOX_ABOUT_INFO_HEADER}"
        "-DAPPBOX_VERSION=${appbox_VERSION}"
        "-DAPPBOX_GIT_EXECUTABLE=${GIT_EXECUTABLE}"
        -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/GenerateAboutInfo.cmake"
    BYPRODUCTS "${APPBOX_ABOUT_INFO_HEADER}"
    COMMENT "Generating the build information header"
    VERBATIM
)

###############################################################################
# Payloads embedded as Windows resources
#
# The packer carries the launcher program as an RCDATA resource of its own
# executable, so a pack run needs no launcher file beside it. The launcher in
# turn carries the two sandbox injection modules as its own RCDATA resources, so
# an extracted archive needs no module file either; those resources belong to the
# launcher and are generated by launcher/CMakeLists.txt from
# cmake/LauncherResources.rc.in.
#
# The payload of the packer is a build product of this build tree, so the
# resource script which names it cannot be checked in: it is generated from
# cmake/EmbeddedResources.rc.in, with the absolute path of the payload
# substituted for its placeholder. src/resource.rc includes the generated
# script, so the payload is compiled into the resource script of the packer.
#
# The resource compiler reads the backslash of a Windows path as an escape
# character inside a string literal, so every separator is doubled before the
# path is substituted.
###############################################################################
set(APPBOX_RESOURCE_LAUNCHER "${CMAKE_BINARY_DIR}/resource/appbox/AppBoxLauncher.exe")

function(appbox_escape_resource_path out_var path)
    file(TO_NATIVE_PATH "${path}" native)
    string(REPLACE "\\" "\\\\" escaped "${native}")
    set(${out_var} "${escaped}" PARENT_SCOPE)
endfunction()

appbox_escape_resource_path(APPBOX_RESOURCE_LAUNCHER_RC "${APPBOX_RESOURCE_LAUNCHER}")

configure_file(
    "${CMAKE_CURRENT_SOURCE_DIR}/cmake/EmbeddedResources.rc.in"
    "${APPBOX_GENERATED_DIR}/EmbeddedResources.generated.rc"
    @ONLY
)

###############################################################################
# Application manifest
#
# resource.rc embeds the wxWidgets application manifest (per monitor v2 DPI
# awareness) through wx/msw/wx.rc. The manifest generated by the linker would be
# a second RT_MANIFEST resource with the same id, which makes cvtres fail with
# CVT1100, so it is turned off here. The wx manifest already carries the
# trustInfo element (level="asInvoker") of the default one.
###############################################################################
if (MSVC)
    target_link_options(AppBox PRIVATE /MANIFEST:NO)
endif ()

# Binary API of Detours, which writes the marker of a launcher into the payload
# image (DetourBinaryOpen and friends). The launcher and the injection modules
# link the same library.
find_package(Detours REQUIRED)
target_link_libraries(AppBox PRIVATE Detours)

find_package(wxWidgets REQUIRED)
target_link_libraries(AppBox PRIVATE wx::core wx::base wx::adv)

# Known folder resolution of `common/KnownFolder.hpp` (SHGetKnownFolderPath and
# CoTaskMemFree), which the packer uses to write the references of the capture.
target_link_libraries(AppBox PRIVATE shell32 ole32)

# Registry access of the packer core, which writes the virtual registry into a
# hive file (RegLoadAppKeyW and friends).
target_link_libraries(AppBox PRIVATE advapi32)

# Version resource of the packer core, which reads the program information the
# launcher payload inherits from the main program of the packaged application
# (GetFileVersionInfoW and VerQueryValueW).
target_link_libraries(AppBox PRIVATE version)

# Trace data helper of Windows, which decodes the payload of the events the
# realtime capture consumes (TdhGetEventInformation and TdhFormatProperty).
target_link_libraries(AppBox PRIVATE tdh)

# zlib must be configured before libzip: the libzip build performs its own
# find_package(ZLIB 1.1.2 REQUIRED) which reuses the already built target.
find_package(ZLIB REQUIRED)
find_package(libzip REQUIRED)
target_link_libraries(AppBox PRIVATE libzip::zip)

# Snapshot files of the capture workspace are SQLite databases, so the core
# links the library of the submodule (see cmake/Findsqlite.cmake).
find_package(sqlite REQUIRED)
target_link_libraries(AppBox PRIVATE sqlite::sqlite3)

###############################################################################
# Embed the payloads
#
# The launcher is the RCDATA resource of this executable (see the `Payloads
# embedded as Windows resources` section above), so a pack run needs no file
# beside the executable. The resource compiler reads the payload from the build
# tree, which is why the target depends on the launcher, which copies itself
# below the build resource tree in its POST_BUILD command. The two sandbox
# injection modules are no dependency of this target: the launcher carries them
# as its own resources, so the builds which produce them belong to the launcher
# and this target reaches them through the dependency below.
###############################################################################
